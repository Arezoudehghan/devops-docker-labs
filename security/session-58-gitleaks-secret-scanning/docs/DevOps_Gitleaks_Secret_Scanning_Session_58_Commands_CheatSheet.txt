# cat > .gitlab-ci.yml <<'EOF'
stages:
  - security
  - build

secret_scan:
  stage: security

  tags:
    - dev-shell

  variables:
    GIT_DEPTH: "0"

  script:
    - gitleaks version
    - gitleaks git --verbose --redact --report-format json --report-path gitleaks-report.json .

  artifacts:
    when: always
    paths:
      - gitleaks-report.json
    expire_in: 1 day

build:
  stage: build

  tags:
    - dev-shell

  script:
    - echo "Secret scan passed"
    - echo "Now Docker build can start"
EOF

Explanation:
Creates `.gitlab-ci.yml` with a security stage that runs Gitleaks before the build stage.

Options:
- > = output redirection: Sends the output of `cat` to `.gitlab-ci.yml`, replacing the file if it already exists.
- <<'EOF' = quoted here-document: Supplies all lines up to the final `EOF` to `cat`; quoting `EOF` prevents shell expansion inside the here-document.
