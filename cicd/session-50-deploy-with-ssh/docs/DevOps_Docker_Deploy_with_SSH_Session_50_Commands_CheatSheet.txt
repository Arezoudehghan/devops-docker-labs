# printf '%s\n' "$NEXUS_PASSWORD" |
ssh \
  -i "$SSH_PRIVATE_KEY" \
  -o BatchMode=yes \
  -o StrictHostKeyChecking=yes \
  "$DEPLOY_USER@$DEPLOY_HOST" \
  "docker login '$NEXUS_REGISTRY' \
  --username '$NEXUS_USER' \
  --password-stdin"

Explanation:
Sends the Nexus password through standard input to an SSH session and runs docker login on the remote deploy server.

Options:
- %s\n = printf format: Prints the password as a string followed by a newline.
- | = pipe operator: Sends printf standard output to ssh standard input, which becomes the remote command's standard input.
- -i "$SSH_PRIVATE_KEY" = identity file: Uses the private key file referenced by SSH_PRIVATE_KEY.
- -o BatchMode=yes = SSH option: Disables interactive password and passphrase prompts.
- -o StrictHostKeyChecking=yes = SSH option: Requires the remote host key to match a trusted known_hosts entry.
- --username '$NEXUS_USER' = username: Supplies the Nexus username to docker login.
- --password-stdin = password from standard input: Reads the registry password from standard input instead of the command line.

# ssh \
  -i "$SSH_PRIVATE_KEY" \
  -o BatchMode=yes \
  -o StrictHostKeyChecking=yes \
  "$DEPLOY_USER@$DEPLOY_HOST" \
  "IMAGE='$IMAGE' \
  CONTAINER_NAME='$CONTAINER_NAME' \
  DEPLOY_PORT='$DEPLOY_PORT' \
  APP_PORT='$APP_PORT' \
  bash -s" <<'REMOTE'

set -euo pipefail

echo "Pulling image: $IMAGE"

docker pull "$IMAGE"

if docker ps -a \
  --format '{{.Names}}' \
  | grep -qx "$CONTAINER_NAME"; then

  echo "Stopping old container..."

  docker stop \
    --timeout 20 \
    "$CONTAINER_NAME"

  echo "Removing old container..."

  docker rm \
    "$CONTAINER_NAME"
fi

echo "Starting new container..."

docker run \
  -d \
  --name "$CONTAINER_NAME" \
  --restart unless-stopped \
  -p "${DEPLOY_PORT}:${APP_PORT}" \
  "$IMAGE"

echo "Container status:"

docker ps \
  --filter "name=^/${CONTAINER_NAME}$"

REMOTE

Explanation:
Technical correction: the lesson used docker stop --time 20; this entry uses the documented docker stop --timeout 20 option. The command connects to DEV-2, passes deployment variables to a remote Bash script, pulls the new image, replaces the old container if present, starts the new container, and shows its status.

Options:
- -i "$SSH_PRIVATE_KEY" = identity file: Uses the selected SSH private key.
- -o BatchMode=yes = SSH option: Disables interactive authentication prompts.
- -o StrictHostKeyChecking=yes = SSH option: Requires strict SSH host-key verification.
- -s in bash -s = read commands from standard input: Makes Bash execute the here-document received on standard input.
- <<'REMOTE' = here-document operator: Sends the enclosed script to the remote Bash process without local shell expansion inside the here-document body.
- -e in set = errexit: Exits the shell when an unhandled command fails.
- -u in set = nounset: Treats use of an unset variable as an error.
- -o pipefail in set = pipefail: Makes a pipeline fail when any command in that pipeline fails.
- -a in docker ps = --all (all): Includes stopped containers.
- --format '{{.Names}}' = format: Prints only container names using the specified Docker Go template.
- | = pipe operator: Sends docker ps output to grep.
- -q in grep = --quiet (quiet): Suppresses normal grep output and uses only the exit status.
- -x in grep = --line-regexp (line regexp): Requires the whole line to match the container name.
- --timeout 20 = timeout: Allows 20 seconds for the container to stop before Docker forces termination.
- -d = --detach (detach): Runs the new container in the background.
- --name "$CONTAINER_NAME" = name: Assigns the requested container name.
- --restart unless-stopped = restart policy: Restarts the container unless it was explicitly stopped.
- -p "${DEPLOY_PORT}:${APP_PORT}" = --publish (publish): Maps the deploy host port to the application port.
- --filter "name=^/${CONTAINER_NAME}$" = filter: Restricts docker ps output to the exact container name pattern.

# set -euo pipefail

Explanation:
Enables stricter Bash error handling so failed commands, unset variables, and failed pipeline elements are not silently ignored.

Options:
- -e = errexit: Exits after an unhandled command failure.
- -u = nounset: Treats an unset variable as an error.
- -o pipefail = pipefail: Makes a pipeline return failure if any command in the pipeline fails.

# for i in $(seq 1 10); do
  if curl \
    --fail \
    --silent \
    --show-error \
    "http://${DEPLOY_HOST}:${DEPLOY_PORT}/"; then

    echo "Application is healthy."
    exit 0
  fi

  echo "Waiting for application..."
  sleep 2
done

echo "Application health check failed."
exit 1

Explanation:
Runs up to ten HTTP health checks against the deployed application, waiting two seconds between failed attempts. It exits successfully on the first healthy response and fails after all attempts are exhausted.

Options:
- $(seq 1 10) = command substitution: Runs seq and substitutes the numbers 1 through 10 into the for loop.
- --fail = fail on HTTP errors: Makes curl return a failure status for HTTP response codes 400 and above.
- --silent = silent: Hides the progress meter and normal status output.
- --show-error = show error: Displays curl error messages even when silent mode is enabled.
- exit 0 = successful shell exit status: Ends the script with success.
- sleep 2 = delay value: Waits two seconds before the next attempt.
- exit 1 = failed shell exit status: Ends the script with failure.
