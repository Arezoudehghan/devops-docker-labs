# cat > compose.yaml <<'EOF'
name: compose-lab37

services:
  db:
    image: postgres:17.11-alpine
    restart: unless-stopped

    environment:
      POSTGRES_DB: ${POSTGRES_DB}
      POSTGRES_USER: ${POSTGRES_USER}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}

    volumes:
      - db_data:/var/lib/postgresql/data

    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U $${POSTGRES_USER} -d $${POSTGRES_DB}"]
      interval: 5s
      timeout: 3s
      retries: 10
      start_period: 10s

    networks:
      - backend

  adminer:
    image: adminer:6.0.1
    restart: unless-stopped

    depends_on:
      db:
        condition: service_healthy
        restart: true

    environment:
      ADMINER_DEFAULT_SERVER: db

    ports:
      - "${ADMINER_PORT}:8080"

    networks:
      - backend

volumes:
  db_data:

networks:
  backend:
    driver: bridge
EOF

Explanation:
Creates the complete compose.yaml file for PostgreSQL and Adminer, including health checking, dependency handling, networking, persistent storage, and restart policies.

Options:
- > = output redirection: Write the command output to compose.yaml, replacing the file if it already exists.
- <<'EOF' = quoted here-document: Pass all lines up to EOF to cat without shell expansion inside the here-document.
- -U = --username (username): In pg_isready, use the value of POSTGRES_USER as the PostgreSQL user name.
- -d = --dbname (database name): In pg_isready, use the value of POSTGRES_DB as the database name or connection string.
- $$ = escaped dollar sign for Compose: Preserve a single dollar sign so the variable is expanded inside the container instead of by Compose.

# pg_isready -U $${POSTGRES_USER} -d $${POSTGRES_DB}

Explanation:
Checks PostgreSQL readiness by using the configured user and database variables inside the container.

Options:
- -U = --username (username): Specify the PostgreSQL user name.
- $${POSTGRES_USER} = Compose-escaped variable: Preserve ${POSTGRES_USER} for expansion inside the container.
- -d = --dbname (database name): Specify the database name or connection string.
- $${POSTGRES_DB} = Compose-escaped variable: Preserve ${POSTGRES_DB} for expansion inside the container.
