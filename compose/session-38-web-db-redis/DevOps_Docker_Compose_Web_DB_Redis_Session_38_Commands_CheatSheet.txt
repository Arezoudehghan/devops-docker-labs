# cat > web/app.py <<'EOF'
import os

import psycopg
import redis
from flask import Flask, jsonify

app = Flask(__name__)

DB_HOST = os.getenv("DB_HOST", "db")
DB_PORT = int(os.getenv("DB_PORT", "5432"))
DB_NAME = os.getenv("DB_NAME", "appdb")
DB_USER = os.getenv("DB_USER", "appuser")
DB_PASSWORD = os.getenv("DB_PASSWORD", "")

REDIS_HOST = os.getenv("REDIS_HOST", "redis")
REDIS_PORT = int(os.getenv("REDIS_PORT", "6379"))

redis_client = redis.Redis(
    host=REDIS_HOST,
    port=REDIS_PORT,
    db=0,
    decode_responses=True,
    socket_connect_timeout=2,
    socket_timeout=2,
)


def get_db_connection():
    return psycopg.connect(
        host=DB_HOST,
        port=DB_PORT,
        dbname=DB_NAME,
        user=DB_USER,
        password=DB_PASSWORD,
        connect_timeout=3,
    )


@app.get("/health")
def health():
    try:
        redis_client.ping()

        with get_db_connection() as conn:
            with conn.cursor() as cur:
                cur.execute("SELECT 1;")
                cur.fetchone()

        return jsonify(
            status="healthy",
            database="ok",
            redis="ok",
        ), 200

    except Exception as exc:
        return jsonify(
            status="unhealthy",
            error=str(exc),
        ), 503


@app.get("/")
def index():
    try:
        redis_views = redis_client.incr("page_views")

        with get_db_connection() as conn:
            with conn.cursor() as cur:
                cur.execute(
                    "INSERT INTO visits (source) VALUES (%s) RETURNING id;",
                    ("web",),
                )

                visit_id = cur.fetchone()[0]

                cur.execute("SELECT COUNT(*) FROM visits;")
                db_visits = cur.fetchone()[0]

        return jsonify(
            message="Session 38 Web + PostgreSQL + Redis is running",
            redis_page_views=redis_views,
            database_total_visits=db_visits,
            last_visit_id=visit_id,
        ), 200

    except Exception as exc:
        return jsonify(
            status="error",
            error=str(exc),
        ), 503
EOF

Options:
- > = redirection: Creates or overwrites web/app.py.
- <<'EOF' = quoted here-document: Sends the block literally to cat until EOF without expanding variables or substitutions.

# cat > web/Dockerfile <<'EOF'
FROM python:3.13.15-slim-bookworm

ENV PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1

WORKDIR /app

COPY requirements.txt .

RUN pip install \
    --no-cache-dir \
    --disable-pip-version-check \
    --trusted-host pypi.org \
    --trusted-host files.pythonhosted.org \
    -r requirements.txt

COPY app.py .

RUN useradd --create-home --uid 10001 appuser \
    && chown -R appuser:appuser /app

USER appuser

EXPOSE 8000

CMD ["gunicorn", "--bind", "0.0.0.0:8000", "--workers", "2", "--threads", "2", "--timeout", "30", "app:app"]
EOF

Explanation:
Creates the Dockerfile for the web service, including dependency installation, a non-root user, and the Gunicorn startup command.

Options:
- > = redirection: Creates or overwrites web/Dockerfile.
- <<'EOF' = quoted here-document: Writes the Dockerfile block literally until EOF.
- --no-cache-dir = pip option: Disables the pip download cache.
- --disable-pip-version-check = pip option: Disables pip's automatic version check.
- --trusted-host pypi.org = pip option: Marks pypi.org as a trusted package host.
- --trusted-host files.pythonhosted.org = pip option: Marks files.pythonhosted.org as a trusted package host.
- -r = --requirement (requirement): Installs packages listed in requirements.txt.
- --create-home = useradd option: Creates a home directory for appuser.
- --uid 10001 = useradd option: Assigns numeric user ID 10001 to appuser.
- && = AND operator: Runs chown only if useradd succeeds.
- -R = --recursive (recursive): Makes chown apply recursively to /app.
- --bind 0.0.0.0:8000 = Gunicorn option: Binds Gunicorn to all container interfaces on port 8000.
- --workers 2 = Gunicorn option: Starts two worker processes.
- --threads 2 = Gunicorn option: Uses two threads per worker.
- --timeout 30 = Gunicorn option: Sets the worker timeout to 30 seconds.

    healthcheck:

        [
          "CMD",
          "python",
          "-c",
          "import urllib.request; urllib.request.urlopen('http://localhost:8000/health', timeout=2)"
        ]
      interval: 10s
      timeout: 3s
      retries: 5
      start_period: 5s

    networks:
      - app_net


  db:
    image: postgres:17.11-alpine

    restart: unless-stopped

    environment:
      POSTGRES_DB: ${POSTGRES_DB}
      POSTGRES_USER: ${POSTGRES_USER}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}

    volumes:
      - postgres_data:/var/lib/postgresql/data
      - ./db/init.sql:/docker-entrypoint-initdb.d/001-init.sql:ro

    healthcheck:
      test:
        [
          "CMD-SHELL",
          "pg_isready -U $${POSTGRES_USER} -d $${POSTGRES_DB}"
        ]
      interval: 5s
      timeout: 3s
      retries: 10
      start_period: 10s

    networks:
      - app_net


  redis:
    image: redis:7.4.10-alpine

    restart: unless-stopped

    command:
      [
        "redis-server",
        "--save",
        "",
        "--appendonly",
        "no"
      ]

    healthcheck:
      test:
        [
          "CMD",
          "redis-cli",
          "ping"
        ]
      interval: 5s
      timeout: 3s
      retries: 5
      start_period: 5s

    networks:
      - app_net


networks:
  app_net:
    driver: bridge


volumes:
  postgres_data:
EOF

Explanation:
Creates the complete Docker Compose file for the web, PostgreSQL, and Redis services.

Options:
- > = redirection: Creates or overwrites compose.yaml.
- <<'EOF' = quoted here-document: Writes the Compose configuration literally until EOF.
- -c = Python command-string option: Executes the following Python code string in the web healthcheck.
- -U = --username (username): pg_isready uses the specified PostgreSQL user from POSTGRES_USER.
- -d = --dbname (database name): pg_isready checks the database named by POSTGRES_DB.
- $$ = Compose dollar escape: Preserves a dollar sign so the container shell can expand POSTGRES_USER and POSTGRES_DB.
- --save "" = redis-server option: Disables snapshot save rules by supplying an empty save configuration.
- --appendonly no = redis-server option: Disables append-only-file persistence.
